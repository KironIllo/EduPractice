@startuml
[*] --> Страница_удаления

Страница_удаления --> Ввод_пароля : открыть страницу
Ввод_пароля --> Проверка_пароля : нажать "Удалить аккаунт"

Проверка_пароля --> Страница_удаления : пароль неверен
Проверка_пароля --> Удаление_аккаунта : пароль верен

Удаление_аккаунта --> [*] : возврат на главную страницу

state Ввод_пароля {
  entry: отобразить поле password
  do: ожидать ввод
  exit: считать пароль
}

state Проверка_пароля {
  entry2: вызвать authenticate()
  do2: сравнить пароль с хэшем в БД
  exit2: вернуть результат
}

state Удаление_аккаунта {
  entry3: выполнить logout
  do3: удалить пользователя
  exit3: очистить сессию
}
@enduml
