@startuml
[*] --> Страница_входа

Страница_входа --> Заполнение_формы : открыть страницу формы
Заполнение_формы --> Проверка_данных : нажать "Войти"

Проверка_данных --> Страница_входа : ошибка валидации / неверный пароль
Проверка_данных --> Успешный_вход : аутентификация пройдена

Успешный_вход --> [*] : редирект на главную страницу

state Заполнение_формы {
  entry: отобразить поля username/password
  do: ожидать ввода
  exit: считать введённые данные
}

state Проверка_данных {
  entry2: вызвать authenticate()
  do2: проверить учётные данные в БД
  exit2: вернуть результат (успех/неуспех)
}
@enduml
