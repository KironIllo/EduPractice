@startuml
class User {
  +username: str
  +password: str
  +email: str
  +last_login: datetime
  +is_active: bool
  +login()
  +logout()
}

class Profile {
  +user: ForeignKey(User)
  +balance: Decimal
  +get_or_create()
}

class Transaction {
  +user: ForeignKey(User)
  +moneyid: str
  +money: str
  +transaction_type: str
  +count: Decimal
  +price: Decimal
  +endprice: Decimal
  +created_at: datetime
}

class TransactionForm {
  +fields
  +is_valid()
  +save()
}

  -smtp_server: str
  -smtp_port: int
  -username: str
  -password: str
  +generate_verification_code()

}

  +get_asset_price(asset_id)
}

  +get_metal_price(metal_code)
  +get_all_metals()

}

  +get_currency_price(currency_id)
}
